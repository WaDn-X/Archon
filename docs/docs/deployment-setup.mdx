# Deployment Setup Guide

This guide covers deploying Zippy Archon in various environments, from local development to production deployment with high availability and monitoring.

## 🚀 Quick Start Deployment

### One-Command Production Launch

```bash
# Clone and setup
git clone https://github.com/ZippyNetworks/Zippy-Archon.git
cd Zippy-Archon

# Configure environment
cp env.production.example .env.production
# Edit .env.production with your production values

# Deploy with production profile
ENVIRONMENT=production ./deploy-production.sh
```

```yaml
version: '3.8'

services:
  postgres:
    image: postgres:15-alpine
    environment:
      POSTGRES_DB: zippy_archon
      POSTGRES_USER: zippy_user
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
    volumes:
      - postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U zippy_user -d zippy_archon"]
      interval: 30s
      timeout: 10s
      retries: 3

  redis:
    image: redis:7-alpine
    command: redis-server --appendonly yes --requirepass ${REDIS_PASSWORD}
    volumes:
      - redis_data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "--raw", "incr", "ping"]
      interval: 30s
      timeout: 10s
      retries: 3

  archon-server:
    build:
      context: ./python
      dockerfile: Dockerfile.server
    environment:
      ENVIRONMENT: production
      DATABASE_URL: postgresql://zippy_user:${POSTGRES_PASSWORD}@postgres:5432/zippy_archon
      REDIS_URL: redis://redis:6379
      REDIS_PASSWORD: ${REDIS_PASSWORD}
      JWT_SECRET_KEY: ${JWT_SECRET_KEY}
      XAI_API_KEY: ${XAI_API_KEY}
    ports:
      - "8181:8181"
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    healthcheck:
      test: ["CMD", "python", "-c", "import requests; requests.get('http://localhost:8181/health')"]
      interval: 30s
      timeout: 10s
      retries: 3

  archon-frontend:
    build: ./archon-ui-main
    ports:
      - "3737:3737"
    environment:
      VITE_API_URL: http://archon-server:8181
```

### Application Metrics

```python
# python/src/server/config/monitoring.py
from prometheus_client import Counter, Histogram, Gauge
import time

# Request metrics
REQUEST_COUNT = Counter(
    'archon_requests_total',
    'Total number of requests',
    ['method', 'endpoint', 'status']
)

REQUEST_LATENCY = Histogram(
    'archon_request_duration_seconds',
    'Request duration in seconds',
    ['method', 'endpoint']
)

# Business metrics
ACTIVE_USERS = Gauge(
    'archon_active_users',
    'Number of active users'
)

PROJECTS_CREATED = Counter(
    'archon_projects_created_total',
    'Total number of projects created'
)

TASKS_COMPLETED = Counter(
    'archon_tasks_completed_total',
    'Total number of tasks completed'
)

# AI metrics
AI_REQUESTS = Counter(
    'archon_ai_requests_total',
    'Total number of AI requests',
    ['provider', 'model']
)

AI_LATENCY = Histogram(
    'archon_ai_request_duration_seconds',
    'AI request duration in seconds',
    ['provider', 'model']
)

def track_request_metrics(method: str, endpoint: str, status: int, duration: float):
    """Track request metrics."""
    REQUEST_COUNT.labels(method=method, endpoint=endpoint, status=status).inc()
    REQUEST_LATENCY.labels(method=method, endpoint=endpoint).observe(duration)
```

## 🔒 Security Configuration

### SSL/TLS Setup

<Tabs>
<TabItem value="nginx" label="Nginx Reverse Proxy">

```nginx
# nginx/nginx.conf
events {
    worker_connections 1024;
}

http {
    upstream archon_backend {
        server archon-server:8181;
    }

    upstream archon_frontend {
        server archon-frontend:3737;
    }

    server {
        listen 80;
        server_name yourdomain.com www.yourdomain.com;

        # Redirect HTTP to HTTPS
        return 301 https://$server_name$request_uri;
    }

    server {
        listen 443 ssl http2;
        server_name yourdomain.com www.yourdomain.com;

        # SSL configuration
        ssl_certificate /etc/nginx/ssl/fullchain.pem;
        ssl_certificate_key /etc/nginx/ssl/privkey.pem;
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers ECDHE-RSA-AES256-GCM-SHA512:DHE-RSA-AES256-GCM-SHA512:ECDHE-RSA-AES256-GCM-SHA384;

        # Security headers
        add_header X-Frame-Options DENY;
        add_header X-Content-Type-Options nosniff;
        add_header X-XSS-Protection "1; mode=block";
        add_header Strict-Transport-Security "max-age=31536000; includeSubDomains";

        # Frontend
        location / {
            proxy_pass http://archon_frontend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        # Backend API
        location /api/ {
            proxy_pass http://archon_backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        # WebSocket support
        location /socket.io/ {
            proxy_pass http://archon_backend;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-Proto $scheme;
        }
    }
}
```

```yaml

    deploy:
      mode: replicated
      replicas: 3
      resources:

          cpus: '1.0'
```

### Load Balancing

```nginx
# Load balancer configuration
upstream backend_servers {
    least_conn;
    server archon-server-1:8181 weight=1;
    server archon-server-2:8181 weight=1;
    server archon-server-3:8181 weight=1;
}

server {
    listen 443 ssl;
    server_name api.yourdomain.com;

    location / {
        proxy_pass http://backend_servers;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # Load balancing health checks
        health_check interval=10s;
    }
}
```

## 🚨 Backup & Recovery

### Automated Backups

```bash
#!/bin/bash
# backup.sh

# Database backup
docker exec postgres pg_dump -U zippy_user zippy_archon > backup_$(date +%Y%m%d_%H%M%S).sql

# Redis backup
docker exec redis redis-cli --raw BGSAVE

# Upload to cloud storage
aws s3 cp backup_*.sql s3://your-backup-bucket/
aws s3 cp dump.rdb s3://your-backup-bucket/redis/

# Cleanup old backups
find . -name "backup_*.sql" -mtime +7 -delete
```

### Disaster Recovery

```bash
#!/bin/bash
# restore.sh

# Stop services
docker-compose down

# Restore database
docker-compose up -d postgres
docker exec -i postgres psql -U zippy_user -d zippy_archon < backup_latest.sql

# Restore Redis
docker-compose up -d redis
docker exec redis redis-cli FLUSHALL
docker exec redis redis-cli < dump.rdb

# Start all services
docker-compose up -d
```
